func _ready() -> void:
	controller = get_parent()
	_build_ui()
	refresh_all()


func _c() -> Node:
	if controller == null:
		controller = get_parent()
	return controller


func _call_arr(method: String, args: Array, default_value):
	var c: = _c()
	if c != null and c.has_method(method):
		return c.callv(method, args)
	return default_value


func _getv(src, keys: Array, default_value):
	if src == null:
		return default_value
	if src is Dictionary:
		for k in keys:
			if src.has(k) and src[k] != null:
				return src[k]
		return default_value
	if src is Object:
		for k in keys:
			var v = src.get(String(k))
			if v != null:
				return v
	return default_value

func _sync_laws() -> void:

	var laws: Dictionary = _call_arr("get_laws", [], {})
	if laws.is_empty():
		return
	if laws.has("minimum_wage"):
		_wage_spin.set_value_no_signal(float(laws["minimum_wage"]))
	if laws.has("income_tax"):
		var pct: = float(laws["income_tax"]) * 100.0
		_tax_slider.set_value_no_signal(pct)
		_tax_value.text = "%d%%" % int(round(pct))


func _on_tax_changed(v: float) -> void:
	_tax_value.text = "%d%%" % int(v)
	_call_arr("do_set_law", ["income_tax", v / 100.0], null)


func _on_publish_need() -> void:
	var need: String = NEED_TYPES[_need_option.selected] if _need_option.selected >= 0 else "grocery"
	var incentive: String = INCENTIVES[_incentive_option.selected] if _incentive_option.selected >= 0 else "none"
	_call_arr("do_publish_need", [need, incentive], null)




func refresh_all() -> void:
	if _day_label == null:
		return
	refresh_top_bar()
	_sync_laws()
	_refresh_town()
	_refresh_requests()
	_refresh_projects()
	_refresh_citizens()


func refresh_top_bar() -> void:
	if _day_label == null:
		return
	var c: = _c()
	var day: = 0
	var date: = ""
	var pop: = 0
	var budget: = 0.0
	var speed: = 0
	if c != null:
		day = int(_getv(c, ["day"], 0))
		if c.has_method("date_text"):
			date = String(c.call("date_text"))
		if c.has_method("get_population"):
			pop = int(c.call("get_population"))
		elif c.has_method("get_citizens"):
			pop = (c.call("get_citizens") as Array).size()
		if c.has_method("get_city_budget"):
			budget = float(c.call("get_city_budget"))
		speed = int(_getv(c, ["speed_index"], 0))
	_day_label.text = "Day %d  %s" % [day, date]
	_pop_label.text = "Population %d" % pop
	_budget_label.text = "City budget $%s" % _money(budget)
	for i in _speed_buttons.size():
		(_speed_buttons[i] as Button).button_pressed = i == speed


func _money(v: float) -> String:
	var neg: = v < 0.0
	var a: = int(absf(v))
	var s: = str(a)
	var out: = ""
	while s.length() > 3:
		out = "," + s.right(3) + out
		s = s.left(s.length() - 3)
	out = s + out
	return ("-" if neg else "") + out


func _refresh_town() -> void:
	var c: = _c()
	if c == null:
		return
	var summary: Dictionary = _getv(c, ["last_summary"], {})
	var econ: Dictionary = _getv(c, ["last_econ_summary"], {})
	var stats: Dictionary = {}
	if c.has_method("compute_town_stats"):
		var st = c.call("compute_town_stats")
		if st is Dictionary:
			stats = st
	var lines: Array = []
	lines.append("Town summary")
	lines.append("Births today: %s" % str(_getv(summary, ["births"], 0)))
	lines.append("Deaths today: %s" % str(_getv(summary, ["deaths"], 0)))
	lines.append("Moved in today: %s" % str(_getv(summary, ["immigrants", "moved_in"], 0)))
	lines.append("Moved out today: %s" % str(_getv(summary, ["emigrants", "moved_out"], 0)))
	lines.append("Employed: %s" % str(_getv(stats, ["employed"], 0)))
	lines.append("Unemployed: %s" % str(_getv(stats, ["unemployed"], 0)))
	lines.append("Businesses open: %s" % str(_getv(stats, ["businesses_open"], _getv(econ, ["businesses_open"], 0))))
	lines.append("Businesses opened today: %s" % str(_getv(econ, ["businesses_started", "opened", "businesses_opened"], 0)))
	lines.append("Businesses closed today: %s" % str(_getv(econ, ["businesses_closed", "closed"], 0)))
	lines.append("Groceries open: %s" % str(_getv(stats, ["grocery_count"], 0)))
	_town_label.text = "\n".join(lines)


func _clear_box(box: VBoxContainer) -> void:
	for child in box.get_children():
		child.queue_free()


func _refresh_requests() -> void:
	_clear_box(_requests_box)
	var petitions: Array = _call_arr("get_petitions", [], [])
	if petitions.is_empty():
		var l: = Label.new()
		l.text = "No petitions yet. Unmet needs create them over time."
		l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		_requests_box.add_child(l)
		return
	for i in petitions.size():
		var p = petitions[i]
		var pid = _getv(p, ["id"], i)
		var title: = String(_getv(p, ["title", "name", "need_type", "need", "what"], "Petition"))
		var sigs = _getv(p, ["signatures", "signature_count", "signers"], 0)
		var status: = String(_getv(p, ["status"], "open"))
		var row: = VBoxContainer.new()
		var l: = Label.new()
		l.text = "%s\nSignatures: %s  Status: %s" % [title, str(sigs), status]
		l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		row.add_child(l)
		var btns: = HBoxContainer.new()
		var approve: = Button.new()
		approve.text = "Approve"
		var deny: = Button.new()
		deny.text = "Deny"
		var id_copy = pid
		approve.pressed.connect(func() -> void: _call_arr("do_respond_petition", [id_copy, true], null))
		deny.pressed.connect(func() -> void: _call_arr("do_respond_petition", [id_copy, false], null))
		btns.add_child(approve)
		btns.add_child(deny)
		row.add_child(btns)
		_requests_box.add_child(row)


func _refresh_projects() -> void:
	_clear_box(_projects_box)
	var projects: Array = _call_arr("get_projects", [], [])
	if projects.is_empty():
		var l: = Label.new()
		l.text = "No projects yet. Start a road project to collect bids."
		l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		_projects_box.add_child(l)
		return
	for i in projects.size():
		var p = projects[i]
		var pid = _getv(p, ["id"], i)
		var title: = String(_getv(p, ["title", "name", "type", "kind"], "Project"))
		var status: = String(_getv(p, ["status"], "bidding"))
		var head: = Label.new()
		head.text = "%s  Status: %s" % [title, status]
		head.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		_projects_box.add_child(head)
		var bids = _getv(p, ["bids"], [])
		if bids is Array:
			for bi in bids.size():
				var b = bids[bi]
				var brow: = HBoxContainer.new()
				var bl: = Label.new()
				bl.text = "%s  $%s  %s days  quality %s" % [
					String(_getv(b, ["company", "bidder", "name"], "Bidder")),
					_money(float(_getv(b, ["price", "cost", "amount"], 0.0))),
					str(_getv(b, ["days", "duration", "duration_days"], "?")),
					str(_getv(b, ["quality_history", "quality", "quality_grade"], "?")),
				]
				bl.size_flags_horizontal = Control.SIZE_EXPAND_FILL
				brow.add_child(bl)
				var award: = Button.new()
				award.text = "Award"
				var pid_copy = pid
				var bi_copy: int = bi
				award.pressed.connect(func() -> void: _call_arr("do_award_project", [pid_copy, bi_copy], null))
				brow.add_child(award)
				_projects_box.add_child(brow)


func _alive_citizens() -> Array:
	var out: Array = []
	for c in _call_arr("get_citizens", [], []):
		if bool(_getv(c, ["alive"], true)) and not bool(_getv(c, ["left_town"], false)):
			out.append(c)
	return out


func _citizen_name(c, fallback: String) -> String:
	var nm: = String(_getv(c, ["name"], ""))
	if nm != "":
		return nm
	var first: = String(_getv(c, ["first_name"], ""))
	var last: = String(_getv(c, ["last_name"], ""))
	var joined: = (first + " " + last).strip_edges()
	return joined if joined != "" else fallback


func _refresh_citizens() -> void:
	var citizens: = _alive_citizens()
	_citizen_list.clear()
	var limit: int = mini(30, citizens.size())
	for i in limit:
		var c = citizens[i]
		var nm: = _citizen_name(c, "Citizen %d" % (i + 1))
		var age = _getv(c, ["age_years", "age"], "?")
		var job: = String(_getv(c, ["job_title", "job", "employer"], "none"))
		if job == "":
			job = "none"
		var cash: = float(_getv(c, ["cash", "money", "balance"], 0.0))
		_citizen_list.add_item("%s, %s, %s, $%s" % [nm, str(age), job, _money(cash)])
	if _selected_citizen >= 0 and _selected_citizen < citizens.size():
		_show_citizen(citizens[_selected_citizen])


func _on_citizen_selected(index: int) -> void:
	_selected_citizen = index
	var citizens: = _alive_citizens()
	if index >= 0 and index < citizens.size():
		_show_citizen(citizens[index])


func _dict_summary(d, keys: Array) -> String:
	if not (d is Dictionary):
		return "none recorded"
	var parts: Array = []
	for k in keys:
		if d.has(k):
			parts.append("%s %s" % [String(k), str(d[k])])
	if parts.is_empty():
		for k in d.keys():
			parts.append("%s %s" % [String(k), str(d[k])])
			if parts.size() >= 6:
				break
	return ", ".join(parts) if not parts.is_empty() else "none recorded"


func _show_citizen(c) -> void:
	var lines: Array = []
	lines.append("Name: %s" % _citizen_name(c, "Unknown"))
	lines.append("Age: %s" % str(_getv(c, ["age_years", "age"], "?")))
	lines.append("Job: %s" % String(_getv(c, ["job_title", "job", "employer"], "none")))
	lines.append("Cash: $%s" % _money(float(_getv(c, ["cash", "money", "balance"], 0.0))))
	lines.append("Credit score: %s" % str(_getv(c, ["credit_score", "credit"], "no history")))
	lines.append("Traits: %s" % _dict_summary(_getv(c, ["traits"], {}), ["ambition", "risk_tolerance", "discipline", "honesty", "sociability", "intelligence", "greed", "kindness"]))
	lines.append("Needs: %s" % _dict_summary(_getv(c, ["needs"], {}), ["hunger", "energy", "shelter", "health", "safety", "money_security", "social", "fun"]))
	lines.append("Recent history:")
	var history = _getv(c, ["history", "history_log", "log"], [])
	if history is Array and not history.is_empty():
		var start: int = maxi(0, history.size() - 5)
		for i in range(start, history.size()):
			var h = history[i]
			if h is Dictionary:
				lines.append("- %s" % String(_getv(h, ["text", "reason", "event"], str(h))))
			else:
				lines.append("- %s" % str(h))
	else:
		lines.append("- nothing yet")
	_citizen_detail.text = "\n".join(lines)
